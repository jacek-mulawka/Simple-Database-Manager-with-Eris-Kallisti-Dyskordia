procedure TCustomSynEdit.DoLinesDeleted(FirstLine, Count: Integer);
begin

  for Mark in Marks do
    //if Mark.Line >= FirstLine + Count then // Jacek Mulawka.
    if Mark.Line > FirstLine + Count then // Jacek Mulawka.
      Mark.Line := Mark.Line - Count
    //else if Mark.Line > FirstLine then // Jacek Mulawka.
    else if Mark.Line > FirstLine + 1 then // Jacek Mulawka.
      Mark.Line := FirstLine;

end;

procedure TCustomSynEdit.DoLinesInserted(FirstLine, Count: Integer);
begin

  for Mark in Marks do
    //if Mark.Line >= FirstLine then // Jacek Mulawka.
    if Mark.Line > FirstLine then // Jacek Mulawka.
      Mark.Line := Mark.Line + Count;

end;
